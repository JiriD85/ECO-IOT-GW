Create new API router following existing patterns (see backup.py for file upload pattern):

```python
from fastapi import APIRouter, HTTPException, UploadFile, File, Depends
from fastapi.responses import Response
from ..services.branding_service import branding_service
from ..models.schemas import BrandingConfig, BrandingStatus
from ..security.auth import get_current_user, require_admin

router = APIRouter(prefix="/branding", tags=["branding"])

@router.get("/config", response_model=BrandingStatus)
async def get_config():
    """Get branding configuration (public for login page)."""
    return branding_service.get_config()

@router.put("/config", response_model=BrandingStatus)
async def update_config(
    config: BrandingConfig,
    current_user = Depends(require_admin)
):
    """Update branding configuration (admin only)."""
    return branding_service.set_config(config)

@router.get("/logo")
async def get_logo():
    """Get logo image (public)."""
    result = branding_service.get_logo()
    if not result:
        raise HTTPException(404, "No logo configured")
    content, content_type = result
    return Response(content=content, media_type=content_type)

@router.post("/logo")
async def upload_logo(
    file: UploadFile = File(...),
    current_user = Depends(require_admin)
):
    """Upload logo (admin only)."""
    content = await file.read()
    branding_service.save_logo(content, file.content_type)
    return {"status": "success"}

@router.delete("/logo")
async def delete_logo(current_user = Depends(require_admin)):
    """Delete logo (admin only)."""
    branding_service.delete_logo()
    return {"status": "success"}

# Same pattern for favicon endpoints
```

Key points:
- GET /config and GET /logo are public (needed for login page)
- All modifications require admin role
- File upload uses UploadFile with await file.read()
- Return Response with correct media_type for images
