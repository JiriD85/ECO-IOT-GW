# =============================================================================
# NTP Models

class NTPServer(BaseModel):
    """NTP server configuration."""
    address: str = Field(..., min_length=1, max_length=255)
    type: str = Field(default="server", pattern="^(server|pool)$")
    options: Optional[str] = None  # e.g., "iburst maxpoll 10"


class NTPConfig(BaseModel):
    """NTP configuration model."""
    servers: List[str] = Field(default_factory=list)
    pools: List[str] = Field(default_factory=list)


class NTPConfigRequest(BaseModel):
    """NTP configuration update request."""
    servers: List[str] = Field(default_factory=list, max_length=10)
    pools: List[str] = Field(default_factory=list, max_length=5)

    @field_validator("servers", "pools")
    @classmethod
    def validate_entries(cls, v):
        for entry in v:
            if not entry or len(entry) > 255:
                raise ValueError("Invalid server/pool address")
        return v


class NTPStatus(BaseModel):
    """NTP synchronization status."""
    synchronized: bool
    reference_id: Optional[str] = None  # e.g., "GPS" or IP
    stratum: Optional[int] = None  # 1-15, lower is better
    ref_time: Optional[datetime] = None  # Last sync time
    system_time: Optional[float] = None  # Offset in seconds
    last_offset: Optional[float] = None  # Last measured offset
    rms_offset: Optional[float] = None  # RMS offset (jitter indicator)
    frequency: Optional[float] = None  # Frequency error in ppm
    root_delay: Optional[float] = None
    root_dispersion: Optional[float] = None
    update_interval: Optional[float] = None
    leap_status: Optional[str] = None


class NTPSource(BaseModel):
    """NTP source status."""
    mode: str  # '^' = server, '=' = peer, '#' = local
    state: str  # '*' = current, '+' = combined, '-' = not combined, '?' = unreachable
    name: str
    stratum: int
    poll: int  # Poll interval in seconds
    reach: str  # Reachability register (octal)
    last_rx: Optional[str] = None  # Time since last response
    last_sample: Optional[str] = None  # Offset info


class TimezoneInfo(BaseModel):
    """Timezone information."""
    current: str
    available: List[str]
```

import logging

from ..models.schemas import (
    ErrorResponse,

    NTPConfigRequest,
    NTPStatus,
    NTPSource,
    TimezoneInfo,
    UserInfo
)
from ..security.auth import get_current_user
from ..services.ntp_service import ntp_service

logger = logging.getLogger(__name__)
router = APIRouter()


def get_client_ip(request: Request) -> str:
    """Extract client IP from request."""
    client_ip = request.headers.get("X-Forwarded-For", "").split(",")[0].strip()
    if not client_ip:
        client_ip = request.client.host if request.client else "unknown"
    return client_ip
```

**Endpoints to implement:**

1. `GET /config` - Get current NTP configuration
   - Returns: NTPConfig (servers, pools)
   - Auth: requires login

2. `PUT /config` - Update NTP configuration
   - Body: NTPConfigRequest
   - Returns: SuccessResponse
   - Auth: requires admin role
   - Audit log: action="ntp_config_update"

3. `GET /status` - Get NTP sync status
   - Returns: NTPStatus
   - Auth: requires login

4. `GET /sources` - Get NTP sources
   - Returns: List[NTPSource]
   - Auth: requires login

5. `GET /timezones` - Get available timezones
   - Returns: TimezoneInfo
   - Auth: requires login

6. `GET /timezone` - Get current timezone
   - Returns: {"timezone": str}
   - Auth: requires login

7. `PUT /timezone` - Set timezone
   - Query param: timezone: str
   - Returns: SuccessResponse
   - Auth: requires login (not admin - anyone can set their preferred timezone view)
   - Audit log: action="timezone_update"
